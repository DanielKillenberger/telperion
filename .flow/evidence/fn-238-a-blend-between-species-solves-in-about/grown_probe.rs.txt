//! fn-238 R1: one grown and one kept prediction of a preset at an age.
use std::time::Instant;
use telperion_core::presets::Preset;
use telperion_space::{expected_counts, expected_kept};

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let preset = match args[1].as_str() {
        "beech" => Preset::EuropeanBeech,
        "oak" => Preset::OregonWhiteOak,
        "spruce" => Preset::NorwaySpruce,
        _ => Preset::Ordinary,
    };
    let age: u32 = args[2].parse().unwrap();
    let mut s = preset.parameters().space.species.unwrap();
    for st in &mut s.states {
        (st.shade_hazard, st.shade_size, st.balance_hazard, st.leaf_girth) = (0.0, 0.0, 0.0, 0.0);
    }
    let t = Instant::now();
    let k = expected_kept(&s, age).unwrap();
    let tk = t.elapsed().as_secs_f64();
    let kp: Vec<usize> = telperion_space::PROBE.iter().map(|a| a.swap(0, std::sync::atomic::Ordering::Relaxed)).collect();
    println!("kept subs {} cells {}", kp[2] / 1_000_000_000, kp[4]);
    let t = Instant::now();
    let c = expected_counts(&s, age).unwrap();
    let tg = t.elapsed().as_secs_f64();
    let g: f64 = (0..s.states.len()).map(|p| c.total(p)).sum();
    let t = Instant::now();
    let l = telperion_space::expected_length(&s, age).unwrap();
    println!("length {l} {:.3}s", t.elapsed().as_secs_f64());
    let pr: Vec<usize> = telperion_space::PROBE.iter().map(|a| a.load(std::sync::atomic::Ordering::Relaxed)).collect();
    let ns: Vec<f64> = telperion_space::NS.iter().map(|a| a.load(std::sync::atomic::Ordering::Relaxed) as f64 * 1e-9).collect();
    println!("calls {} subs {} events {} cells {} | events_s {:.3} shift_s {:.3} clone_s {:.3}", pr[0], pr[1], pr[2], pr[3], ns[0], ns[1], ns[2]);
    println!("{} {age} kept {k:?} {tk:.3}s grown {g} {tg:.3}s", args[1]);
}
