#!/bin/bash
# The strip frames whose factor moved (factors-*.txt), before and after, seed 1.
B=/tmp/claude-1000/-home-daniel-Projects-telperion/dd1a391e-8087-41d5-8d26-b3ea78d9cb37/scratchpad/fn238-base
N=/home/daniel/Projects/telperion/.worktrees/blendcost
R=$N/.flow/evidence/fn-238-a-blend-between-species-solves-in-about/raw/strips
export RAYON_NUM_THREADS=4
while read from to age t; do
  for side in before after; do
    root=$N; [ $side = before ] && root=$B
    mkdir -p $R/$side-$age
    $root/target/ci/examples/space_walk $R/$side-$age $from $to $t 1 $age > $R/$side-$age/log-$from-$to-$t.txt 2>&1 || echo "FAILED $side $from $to $age $t"
  done
  f=walk-$from-$to-$(printf %.3f $t).png
  if cmp -s $R/before-$age/$f $R/after-$age/$f; then echo "$age $f identical"; else echo "$age $f differs: $(compare -metric AE $R/before-$age/$f $R/after-$age/$f null: 2>&1) px"; fi
done <<LIST
beech oak 40 0.25
beech spruce 40 0.25
beech spruce 40 0.5
spruce ordinary 40 0.75
beech spruce 80 0.5
beech spruce 80 0.75
spruce ordinary 80 0.25
spruce ordinary 80 0.5
spruce ordinary 80 0.75
LIST
