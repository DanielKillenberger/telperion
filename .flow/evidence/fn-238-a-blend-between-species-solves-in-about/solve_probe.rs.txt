//! fn-238 R1: for each walk frame of fn-230's strips, the solve's wall time
//! and its predictions (kept and grown: how many, and their seconds).
use std::time::Instant;
use telperion_core::{blend, presets::Preset};

fn main() {
    let names = [
        (Preset::EuropeanBeech, "beech"),
        (Preset::OregonWhiteOak, "oak"),
        (Preset::NorwaySpruce, "spruce"),
        (Preset::Ordinary, "ordinary"),
    ];
    let ages: Vec<f64> = std::env::args().skip(1).map(|a| a.parse().unwrap()).collect();
    println!("from to age t total_s kept_n kept_s grown_n grown_s grown_max_s");
    for &age in &ages {
        for (i, &(p, pn)) in names.iter().enumerate() {
            for &(q, qn) in &names[i + 1..] {
                let (mut a, mut b) = (p.parameters(), q.parameters());
                (a.age, b.age) = (age, age);
                for t in [0.25, 0.5, 0.75] {
                    blend::size::PROBE.lock().unwrap().clear();
                    let s = Instant::now();
                    let _ = blend::families(&a, &b, t);
                    let total = s.elapsed().as_secs_f64();
                    let log = blend::size::PROBE.lock().unwrap().clone();
                    let of = |n: &str| log.iter().filter(|(k, _)| *k == n).map(|(_, s)| *s).collect::<Vec<_>>();
                    let (k, g) = (of("kept"), of("grown"));
                    let gmax = g.iter().cloned().fold(0.0, f64::max);
                    println!(
                        "{pn} {qn} {age} {t} {total:.3} {} {:.3} {} {:.3} {gmax:.3}",
                        k.len(), k.iter().sum::<f64>(), g.len(), g.iter().sum::<f64>()
                    );
                }
            }
        }
    }
}
