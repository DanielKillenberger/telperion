//! fn-238 R3: the seconds `blend::families` takes on each walk frame of
//! fn-230's strips, the frames of a walk in order (0.25, 0.5, 0.75) in one
//! process, as a harness walks them. Run it once per sample in a fresh
//! process, so the first frame of each walk predicts its ends.
use std::time::Instant;
use telperion_core::{blend, presets::Preset};

fn main() {
    let names = [
        (Preset::EuropeanBeech, "beech"),
        (Preset::OregonWhiteOak, "oak"),
        (Preset::NorwaySpruce, "spruce"),
        (Preset::Ordinary, "ordinary"),
    ];
    for age in std::env::args().skip(1).map(|a| a.parse::<f64>().unwrap()) {
        for (i, &(p, pn)) in names.iter().enumerate() {
            for &(q, qn) in &names[i + 1..] {
                let (mut a, mut b) = (p.parameters(), q.parameters());
                (a.age, b.age) = (age, age);
                for t in [0.25, 0.5, 0.75] {
                    let s = Instant::now();
                    blend::families(&a, &b, t).expect("the frame builds");
                    println!("{pn} {qn} {age} {t} {:.3}", s.elapsed().as_secs_f64());
                }
            }
        }
    }
}
