//! fn-238 R3: the lateral factor every walk frame is solved to, as bits,
//! for the walk digest's frames (`catalogue_identity.rs`) and fn-230's
//! strips at 40 and 80, so two builds' factors can be compared.
use telperion_core::{blend, presets::Preset, Family};

fn factor(walked: &Family, a: &Family, b: &Family, t: f64) -> Option<f64> {
    let (sa, sb) = (a.space.species.as_ref()?, b.space.species.as_ref()?);
    let mixed = telperion_space::blend(sa, sb, t).ok()?;
    let sized = walked.space.species.as_ref()?;
    let pairs = sized.states.iter().zip(&mixed.states).flat_map(|(s, m)| {
        s.zones.iter().zip(&m.zones).flat_map(|(p, q)| p.lateral.iter().zip(&q.lateral))
    });
    pairs.filter(|(_, q)| **q > 0.0).map(|(p, q)| p / q).next()
}

fn frame(name: &str, a: &Family, b: &Family, t: f64) {
    let line = match blend::families(a, b, t) {
        Ok(w) => match factor(&w, a, b, t) {
            Some(f) => format!("{f:e} {:016x}", f.to_bits()),
            None => "no-factor".into(),
        },
        Err(e) => format!("refused {e:?}"),
    };
    println!("{name} {t} {line}");
}

fn main() {
    let which = std::env::args().nth(1).unwrap_or_default();
    let presets = [
        Preset::Ordinary,
        Preset::OregonWhiteOak,
        Preset::NorwaySpruce,
        Preset::EuropeanBeech,
        Preset::SilverBirch,
        Preset::DatePalm,
        Preset::Telperion,
        Preset::Laurelin,
    ];
    if which == "digest" {
        let mut ends: Vec<(String, Family)> =
            presets.iter().map(|p| (format!("{p:?}"), p.parameters())).collect();
        let mut stated = Preset::OregonWhiteOak.parameters();
        let g = &mut stated.skeleton.growth;
        (g.influence_radius, g.kill_distance, g.step_distance) = (Some(3.0), Some(0.4), Some(0.3));
        (g.trunk_height, g.max_turn_per_step, g.max_nodes) = (Some(2.5), Some(20.0), Some(90_000));
        let mut half = Preset::NorwaySpruce.parameters();
        half.skeleton.growth.kill_distance = Some(0.2);
        half.skeleton.growth.max_nodes = Some(120_000);
        ends.push(("stated-oak".into(), stated));
        ends.push(("half-spruce".into(), half));
        ends.push(("birch".into(), Preset::SilverBirch.parameters()));
        for (i, (an, a)) in ends.iter().enumerate() {
            for (bn, b) in &ends[i + 1..] {
                for t in [0.13, 0.5, 0.731] {
                    frame(&format!("digest {an} {bn}"), a, b, t);
                }
            }
        }
        return;
    }
    let names = [
        (Preset::EuropeanBeech, "beech"),
        (Preset::OregonWhiteOak, "oak"),
        (Preset::NorwaySpruce, "spruce"),
        (Preset::Ordinary, "ordinary"),
    ];
    let age: f64 = which.parse().expect("an age, or digest");
    for (i, &(p, pn)) in names.iter().enumerate() {
        for &(q, qn) in &names[i + 1..] {
            let (mut a, mut b) = (p.parameters(), q.parameters());
            (a.age, b.age) = (age, age);
            for t in [0.25, 0.5, 0.75] {
                frame(&format!("strip {pn} {qn} {age}"), &a, &b, t);
            }
        }
    }
}
