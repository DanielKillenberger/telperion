mod walk;
use walk::{crossing, made, settings, walk as steps_of};

#[test]
fn probe() {
    for s in settings().iter().filter(|s| s.name.contains("lifespan")) {
        for seed in 1..=3 {
            let steps = steps_of(s, seed);
            let Some(step) = steps.iter().max_by(|a, b| a.made.total_cmp(&b.made)) else { continue };
            if step.made == 0.0 { println!("{} seed {seed}: none made", s.name); continue; }
            let (a, b) = (s.shape(step.from, seed), s.shape(step.to, seed));
            let (_, lineage) = made(&a, &b);
            let wood = crossing(s, seed, step.from, step.to, lineage.unwrap(), 48);
            println!("{} seed {seed} at {:.4}: {wood:e}", s.name, step.to);
        }
    }
}
