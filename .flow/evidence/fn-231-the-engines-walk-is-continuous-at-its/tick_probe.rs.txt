use telperion_core::presets::Preset;
use telperion_space::{expected_counts, expected_kept, expected_length, Species};
fn neutral(s: &Species) -> Species {
    let mut n = s.clone();
    for s in &mut n.states { (s.shade_hazard, s.shade_size, s.balance_hazard, s.leaf_girth) = (0.0, 0.0, 0.0, 0.0); }
    n
}
fn main() {
    let tick = 1.0 / f64::from(1u32 << 20);
    for (name, preset, pa, sign) in [("beech age 9 up", Preset::EuropeanBeech, 9usize, 1.0), ("oak age 2 down", Preset::OregonWhiteOak, 2, -1.0), ("oak age 8 up", Preset::OregonWhiteOak, 8, 1.0), ("spruce age 12 up", Preset::NorwaySpruce, 12, 1.0)] {
        let base = neutral(preset.parameters().space.species.as_ref().unwrap());
        println!("{name}: lifespan {}", base.states[pa].lifespan);
        for k in [0.0, 1.0, 2.0, 3.0, 16.0, 1024.0, 65536.0, 524288.0] {
            let mut s = base.clone();
            s.states[pa].lifespan += sign * k * tick;
            let kept = expected_kept(&s, 40).unwrap().unwrap();
            let c = expected_counts(&s, 40).unwrap();
            let grown: f64 = (0..s.states.len()).map(|p| c.total(p)).sum();
            let wood = expected_length(&s, 40).unwrap();
            println!("  {k:>8} ticks: kept {kept:.0} grown {grown:.0} wood {wood:.1}");
        }
    }
}
